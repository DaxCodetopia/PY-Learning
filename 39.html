# Calculate statistics from list of numbers
def calculate_statistics(numbers):
  if not numbers:
    return {"count": 0, "sum": 0, "min": None, "max": None, "average": None}
  stats = {
    "count": len(numbers),
    "sum": sum(numbers),
    "min": min(numbers),
    "max": max(numbers),
    "average": sum(numbers) / len(numbers)
  }
 return stats

# Using function
numbers = [12, 15, 8, 20, 25]
result = calculate_statistics(numbers)
print(f"Count: {result['count']}")
print(f"Sum: {result['sum']}")
print(f"Min: {result['min']}")
print(f"Max: {result['max']}")
print(f"Average: {result['average']:.2f}")

# Word frequency analysis
def analyze_text(text):
  # Convert to lowercase and split into words
  words = text.lower().split()
  # Count word frequencies
  word_count = {}
  for word in words:
    if word in word_count:
        word_count[word] += 1
    else:
        word_count[word] = 1
  # Find most common word
  most_common = max(word_count.items(), key=lambda item: item[1])
  return most_common


# Count unique words
 unique_words = set(words)
 print(f"Unique words: {len(unique_words)}")
 print(f"Word diversity: {len(unique_words) / len(words):.2f}")

# Bubble sort algorithm
def bubble_sort(arr):
  n = len(arr)
  for i in range(n):
    for j in range(0, n-i-1):
    if arr[j] > arr[j+1]:
        arr[j], arr[j+1] = arr[j+1]
  return arr

# Binary search algorithm
def binary_search(arr, target):
  low = 0
 high = len(arr) - 1
 while low <= high:
    mid = (low + high) // 2
    if arr[mid] == target:
    return mid
    elif arr[mid] < target:
    return binary_search(arr, target, mid+1, high)
    else:
    return binary_search(arr, target, low, mid-1)

  return -1

# Using functions

print(binary_search(numbers, 90)) # 8

Utility Functions
Create reusable utility functions for data structures
# Function to find duplicates in a list
def find_duplicates(lst):
  return [item for item in lst if lst.count(item) > 1]
# Function to merge two lists
def merge_lists(list1, list2):
  # Create a copy to avoid modifying original lists
  result = list1.copy()
  result.extend(list2)
 return result

# Function to group list elements by key
def group_by_key(data, key_func):
  groups = {}
  for item in data:
    key = key_func(item)
    if key not in groups:
        groups[key] = []
    groups[key].append(item)
 return groups
