Practical Examples
# Combining operators
age = 25
has_license = True
has_car = False

# Complex conditions
can_drive = age >= 18 and has_license
print(can_drive) # True

can_drive_own_car = can_drive and has_car
print(can_drive_own_car) # False

# Using parentheses to override precedence
result = not (age < 18 or not has_license)
print(result) # True

# Chaining comparisons
print(10 < age < 30) # True (10 < 25 and 25 < 30)
