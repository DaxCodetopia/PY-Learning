        // --- Interaction Logic ---
        function toggleSubject(btn) {
            // Close other menus (accordion behavior)
            const allMenus = document.querySelectorAll('.submenu');
            const allTitles = document.querySelectorAll('.subject-title');
            
            // Find the sibling UL
            const submenu = btn.nextElementSibling;

            if (submenu.classList.contains('open')) {
                submenu.classList.remove('open');
                btn.classList.remove('active');
            } else {
                // Optional: Close others
                allMenus.forEach(m => m.classList.remove('open'));
                allTitles.forEach(t => t.classList.remove('active'));
                
                submenu.classList.add('open');
                btn.classList.add('active');
            }
        }

        function loadProgram(subject, programData, listElement) {

        // --- Simple Python Syntax Highlighter (Regex) ---
        function highlightPython(code) {
            // Escape HTML entities first to prevent injection/rendering issues
            let safeCode = code.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");

            // Order matters for regex to avoid conflicts
            
            // Comments
            safeCode = safeCode.replace(/(#.*)/g, '<span class="token-comment">$1</span>');
            
            // Strings (single or double quotes)
            safeCode = safeCode.replace(/(["'])(?:(?=(\\?))\2.)*?\1/g, '<span class="token-string">$&</span>');
            
            // Keywords
            const keywords = /\b(def|class|return|if|elif|else|for|while|in|try|except|import|from|as|pass|break|continue|print|True|False|None)\b/g;
            safeCode = safeCode.replace(keywords, '<span class="token-keyword">$1</span>');
            
            // Function Calls (name followed by parentheses) - simple heuristic
            safeCode = safeCode.replace(/\b([a-zA-Z_][a-zA-Z0-9_]*)(?=\()/g, '<span class="token-function">$1</span>');

            // Numbers
            safeCode = safeCode.replace(/\b(\d+)\b/g, '<span class="token-number">$1</span>');

            return safeCode;
        }

        // --- Utility: Copy Code ---
        function copyCode() {
            // Get raw text (strip HTML tags)

            navigator.clipboard.writeText(rawCode).then(() => {

            }).catch(err => {
                console.error('Failed to copy text: ', err);
            });
        }

        function showToast() {

            }, 3000);
        }
