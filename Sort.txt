        <section id="intro">
            <h1>Python Sorting Guide</h1>
            <p class="lead">Sorting is a fundamental operation in programming. In Python, you have powerful built-in tools to order data efficiently and readable.</p>
            <p>This guide covers the two primary ways to sort data: the <code>.sort()</code> method and the <code>sorted()</code> function, and how to use them on lists, tuples, sets, and dictionaries.</p>
        </section>

        <!-- Core Methods -->
        <h2 id="method-sort">1. The <code>.sort()</code> Method</h2>
        <p>The <code>.sort()</code> method modifies the list <strong>in-place</strong>. This means it changes the original list directly and returns <code>None</code>.</p>
        
        <div class="code-window">
            <div class="code-header">
                <div class="window-controls"><div class="control red"></div><div class="control yellow"></div><div class="control green"></div></div>
                <span class="lang-label">python</span>
                <button class="copy-btn" onclick="copyCode(this)">Copy</button>
            </div>
            <pre>
<span class="kw">numbers</span> = [<span class="num">5</span>, <span class="num">2</span>, <span class="num">9</span>, <span class="num">1</span>, <span class="num">5</span>, <span class="num">6</span>]

<span class="com"># Modifies 'numbers' directly</span>
<span class="kw">numbers</span>.<span class="func">sort</span>()

<span class="func">print</span>(<span class="kw">numbers</span>)
<span class="com"># Output: [1, 2, 5, 5, 6, 9]</span></pre>
        </div>

        <div class="note-box">
            <span class="note-title">⚠️ Important</span>
            Because <code>.sort()</code> returns <code>None</code>, assigning it to a variable will result in that variable becoming <code>None</code>. 
            <br><code>new_list = numbers.sort()</code> <span style="color:red"># Avoid this!</span>
        </div>

        <h2 id="method-sorted">2. The <code>sorted()</code> Function</h2>
        <p>The <code>sorted()</code> function builds a <strong>new sorted list</strong> and leaves the original list unchanged. This is useful when you need to keep the original data order.</p>

        <div class="code-window">
            <div class="code-header">
                <div class="window-controls"><div class="control red"></div><div class="control yellow"></div><div class="control green"></div></div>
                <span class="lang-label">python</span>
                <button class="copy-btn" onclick="copyCode(this)">Copy</button>
            </div>
            <pre>
<span class="kw">numbers</span> = [<span class="num">5</span>, <span class="num">2</span>, <span class="num">9</span>, <span class="num">1</span>, <span class="num">5</span>, <span class="num">6</span>]

<span class="com"># Returns a new list, 'numbers' remains unchanged</span>
<span class="kw">sorted_numbers</span> = <span class="func">sorted</span>(<span class="kw">numbers</span>)

<span class="func">print</span>(<span class="str">"Original:"</span>, <span class="kw">numbers</span>)
<span class="func">print</span>(<span class="str">"Sorted:"</span>, <span class="kw">sorted_numbers</span>)</pre>
        </div>

        <!-- Direction -->
        <h2 id="ascending">3. Ascending Order</h2>
        <p>By default, both <code>.sort()</code> and <code>sorted()</code> sort data in <strong>ascending</strong> order (smallest to largest, A to Z). You don't need to pass any extra arguments.</p>

        <div class="code-window">
            <div class="code-header">
                <div class="window-controls"><div class="control red"></div><div class="control yellow"></div><div class="control green"></div></div>
                <span class="lang-label">python</span>
                <button class="copy-btn" onclick="copyCode(this)">Copy</button>
            </div>
            <pre>
<span class="kw">letters</span> = [<span class="str">'d'</span>, <span class="str">'a'</span>, <span class="str">'c'</span>, <span class="str">'b'</span>]
<span class="kw">letters</span>.<span class="func">sort</span>() 
<span class="com"># Result: ['a', 'b', 'c', 'd']</span></pre>
        </div>

        <h2 id="descending">4. Descending Order</h2>
        <p>To sort in <strong>descending</strong> order (largest to smallest, Z to A), set the <code>reverse</code> parameter to <code>True</code>.</p>

        <div class="code-window">
            <div class="code-header">
                <div class="window-controls"><div class="control red"></div><div class="control yellow"></div><div class="control green"></div></div>
                <span class="lang-label">python</span>
                <button class="copy-btn" onclick="copyCode(this)">Copy</button>
            </div>
            <pre>
<span class="kw">numbers</span> = [<span class="num">1</span>, <span class="num">2</span>, <span class="num">3</span>, <span class="num">4</span>, <span class="num">5</span>]

<span class="com"># Method 1: Using .sort()</span>
<span class="kw">numbers</span>.<span class="func">sort</span>(<span class="kw">reverse</span>=<span class="kw">True</span>)

<span class="com"># Method 2: Using sorted()</span>
<span class="kw">desc_numbers</span> = <span class="func">sorted</span>(<span class="kw">numbers</span>, <span class="kw">reverse</span>=<span class="kw">True</span>)

<span class="func">print</span>(<span class="kw">desc_numbers</span>)
<span class="com"># Output: [5, 4, 3, 2, 1]</span></pre>
        </div>

        <!-- Data Structures -->
        <h2 id="sort-list">5. Sorting Lists</h2>
        <p>Lists are the most common data structure to sort. Both methods work directly on lists.</p>
        <ul>
            <li>Use <code>.sort()</code> if you don't need the original order anymore (saves memory).</li>
            <li>Use <code>sorted()</code> if you need to preserve the original list.</li>
        </ul>

        <h2 id="sort-tuple">6. Sorting Tuples</h2>
        <p>Tuples are <strong>immutable</strong> (they cannot be changed). Therefore, they do <strong>not</strong> have a <code>.sort()</code> method. You must use the <code>sorted()</code> function, which will return a new <strong>list</strong>.</p>

        <div class="code-window">
            <div class="code-header">
                <div class="window-controls"><div class="control red"></div><div class="control yellow"></div><div class="control green"></div></div>
                <span class="lang-label">python</span>
                <button class="copy-btn" onclick="copyCode(this)">Copy</button>
            </div>
            <pre>
<span class="kw">my_tuple</span> = (<span class="num">3</span>, <span class="num">1</span>, <span class="num">4</span>, <span class="num">1</span>, <span class="num">5</span>)

<span class="com"># Convert to sorted list</span>
<span class="kw">sorted_list</span> = <span class="func">sorted</span>(<span class="kw">my_tuple</span>)

<span class="func">print</span>(<span class="kw">sorted_list</span>)
<span class="com"># Output: [1, 1, 3, 4, 5] (Note: It is a list, not a tuple)</span>

<span class="com"># Convert back to tuple if needed</span>
<span class="kw">sorted_tuple</span> = <span class="func">tuple</span>(<span class="kw">sorted_list</span>)</pre>
        </div>

        <h2 id="sort-set">7. Sorting Sets</h2>
        <p>Sets are unordered collections. Like tuples, they do not have a <code>.sort()</code> method. Use <code>sorted()</code> to return a sorted list of the set's elements.</p>

        <div class="code-window">
            <div class="code-header">
                <div class="window-controls"><div class="control red"></div><div class="control yellow"></div><div class="control green"></div></div>
                <span class="lang-label">python</span>
                <button class="copy-btn" onclick="copyCode(this)">Copy</button>
            </div>
            <pre>
<span class="kw">my_set</span> = {<span class="num">10</span>, <span class="num">5</span>, <span class="num">20</span>, <span class="num">2</span>}

<span class="kw">sorted_set</span> = <span class="func">sorted</span>(<span class="kw">my_set</span>)

<span class="func">print</span>(<span class="kw">sorted_set</span>)
<span class="com"># Output: [2, 5, 10, 20]</span></pre>
        </div>

        <h2 id="sort-dict">8. Sorting Dictionaries</h2>
        <p>Sorting dictionaries requires deciding whether to sort by <strong>keys</strong> or by <strong>values</strong>. In Python 3.7+, dictionaries maintain insertion order.</p>

        <h3>Sorting by Keys</h3>
        <div class="code-window">

<span class="com"># Get sorted list of keys</span>
<span class="kw">sorted_keys</span> = <span class="func">sorted</span>(<span class="kw">prices</span>)

<span class="func">print</span>(<span class="kw">sorted_keys</span>)
<span class="com"># Output: ['apple', 'banana', 'cherry']</span></pre>
        </div>

        <h3>Sorting by Values</h3>
        <p>To sort by values, we use the <code>key</code> parameter (explained in detail below).</p>
        <div class="code-window">
            <div class="code-header">
                <div class="window-controls"><div class="control red"></div><div class="control yellow"></div><div class="control green"></div></div>

<span class="com"># Sort keys based on their values</span>

        <!-- Advanced -->
        <h2 id="sort-by-key">9. Sorting with the <code>key</code> Parameter</h2>
        <p>The <code>key</code> parameter is powerful. It accepts a function that transforms each element before comparison. It does not change the data, only how it is compared.</p>
        <p>We often use <strong>lambda functions</strong> here.</p>

        <div class="code-window">
            <div class="code-header">
                <div class="window-controls"><div class="control red"></div><div class="control yellow"></div><div class="control green"></div></div>
                <span class="lang-label">python</span>
                <button class="copy-btn" onclick="copyCode(this)">Copy</button>
            </div>
            <pre>
<span class="com"># Sort a list of strings by their length</span>
<span class="kw">words</span> = [<span class="str">"apple"</span>, <span class="str">"pie"</span>, <span class="str">"banana"</span>, <span class="str">"kiwi"</span>]

<span class="kw">words</span>.<span class="func">sort</span>(<span class="kw">key</span>=<span class="kw">len</span>)

<span class="func">print</span>(<span class="kw">words</span>)
<span class="com"># Output: ['pie', 'kiwi', 'apple', 'banana']</span></pre>
        </div>

        <h2 id="real-world">10. Real World Example: Sort by Price/Name</h2>
        <p>A common scenario is sorting a list of objects (dictionaries) representing products. We want to sort by specific fields like <strong>price</strong> or <strong>name</strong>.</p>

<script>
    // --- Dark Mode Logic ---
    const toggleBtn = document.getElementById('theme-toggle');
    const body = document.body;
    const sunIcon = document.getElementById('icon-sun');
    const moonIcon = document.getElementById('icon-moon');

    // Check for saved user preference
    const savedTheme = localStorage.getItem('theme');
    if (savedTheme === 'dark') {
        enableDarkMode();
    }

    toggleBtn.addEventListener('click', () => {
        if (body.classList.contains('dark-mode')) {
            disableDarkMode();
        } else {
            enableDarkMode();
        }
    });

    function enableDarkMode() {
        body.classList.add('dark-mode');
        sunIcon.style.display = 'block';
        moonIcon.style.display = 'none';
        localStorage.setItem('theme', 'dark');
    }

    function disableDarkMode() {
        body.classList.remove('dark-mode');
        sunIcon.style.display = 'none';
        moonIcon.style.display = 'block';
        localStorage.setItem('theme', 'light');
    }

    // --- Copy Code Logic ---
    function copyCode(button) {
        const codeWindow = button.closest('.code-window');
        const codeBlock = codeWindow.querySelector('pre');
        const codeText = codeBlock.innerText;

        navigator.clipboard.writeText(codeText).then(() => {
            const originalText = button.innerText;
            button.innerText = 'Copied!';
            button.style.borderColor = '#27c93f';
            button.style.color = '#27c93f';
            
            setTimeout(() => {
                button.innerText = originalText;
                button.style.borderColor = '#5c6370';
                button.style.color = '#9da5b4';
            }, 2000);
        }).catch(err => {
            console.error('Failed to copy text: ', err);
        });
    }

    // --- Active Link Logic ---
    const sections = document.querySelectorAll('section, h2');
    const navLinks = document.querySelectorAll('.nav-links a');

    window.addEventListener('scroll', () => {
        let current = '';
        sections.forEach(section => {
            const sectionTop = section.offsetTop;
            const sectionHeight = section.clientHeight;
            if (pageYOffset >= (sectionTop - 150)) {
                current = section.getAttribute('id');
            }
        });

        navLinks.forEach(link => {
            link.classList.remove('active');
            if (link.getAttribute('href').includes(current)) {
                link.classList.add('active');
            }
        });
    });
</script>
