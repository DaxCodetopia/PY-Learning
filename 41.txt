                    <div class="card-content">
                        <div class="exercise-desc">
                            Choose the right algorithm for the task. Optimize for speed or memory based on dataset size.
                        </div>
                        <div class="code-example">
                            <span class="comment"># Binary Search: O(log n) - Efficient for sorted data</span><br>
                            <span class="keyword">def</span> <span class="function">binary_search</span>(arr, target):<br>
                            &nbsp;&nbsp;left, right = <span class="number">0</span>, <span class="function">len</span>(arr) - <span class="number">1</span><br>
                            &nbsp;&nbsp;<span class="keyword">while</span> left <= right:<br>
                            &nbsp;&nbsp;&nbsp;&nbsp;mid = (left + right) // <span class="number">2</span><br>
                            &nbsp;&nbsp;&nbsp;&nbsp;<span class="keyword">if</span> arr[mid] == target:<br>
                            &nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;<span class="keyword">return</span> mid<br>
                            &nbsp;&nbsp;&nbsp;&nbsp;<span class="keyword">elif</span> arr[mid] < target:<br>
                            &nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;left = mid + <span class="number">1</span><br>
                            &nbsp;&nbsp;&nbsp;&nbsp;<span class="keyword">else</span>:<br>
                            &nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;right = mid - <span class="number">1</span><br>
                            &nbsp;&nbsp;<span class="keyword">return</span> -<span class="number">1</span>
                        </div>
                        <div class="key-concepts">
                            <div class="concept-tag">binary search</div>
                            <div class="concept-tag">sorting</div>
                            <div class="concept-tag">efficiency</div>
                        </div>
                    </div>

                    <div class="card-header">
                        <i class="material-icons card-icon">storage</i>
                        <div class="card-title">Data Structures</div>
                    </div>
                    <div class="card-content">
                        <div class="exercise-desc">
                            Select structures based on access patterns. Use Sets for O(1) lookups, Lists for order.
                        </div>
                        <div class="code-example">
                            <span class="comment"># List Lookup: O(n) - Slow for large lists</span><br>
                            users_list = [<span class="string">"alice"</span>, <span class="string">"bob"</span>, <span class="string">"charlie"</span>]<br>
                            <span class="keyword">if</span> <span class="string">"dave"</span> <span class="keyword">in</span> users_list:  <span class="comment"># Scans whole list</span><br>
                            &nbsp;&nbsp;<span class="function">print</span>(<span class="string">"Found"</span>)<br><br>
                            
                            <span class="comment"># Set Lookup: O(1) - Instant</span><br>
                            users_set = {<span class="string">"alice"</span>, <span class="string">"bob"</span>, <span class="string">"charlie"</span>}<br>
                            <span class="keyword">if</span> <span class="string">"dave"</span> <span class="keyword">in</span> users_set:  <span class="comment"># Hash map access</span><br>
                            &nbsp;&nbsp;<span class="function">print</span>(<span class="string">"Found"</span>)<br><br>
                            
                            <span class="comment"># Queue Implementation</span><br>
                            <span class="keyword">from</span> collections <span class="keyword">import</span> deque<br>
                            queue = <span class="function">deque</span>([<span class="number">1</span>, <span class="number">2</span>, <span class="number">3</span>])<br>
                            queue.<span class="function">append</span>(<span class="number">4</span>)  <span class="comment"># Efficient O(1)</span>
                        </div>
                        <div class="key-concepts">
                            <div class="concept-tag">sets vs lists</div>

                        <div class="exercise-desc">
                            Analyze how runtime scales with input size. Avoid nested loops when possible.
                        </div>
                        <div class="code-example">
                            <span class="comment"># O(n^2) - Quadratic Time (Slow)</span><br>
                            <span class="keyword">def</span> <span class="function">find_duplicates_slow</span>(arr):<br>
                            &nbsp;&nbsp;<span class="keyword">for</span> i <span class="keyword">in</span> <span class="function">range</span>(<span class="function">len</span>(arr)):<br>
                            &nbsp;&nbsp;&nbsp;&nbsp;<span class="keyword">for</span> j <span class="keyword">in</span> <span class="function">range</span>(i + <span class="number">1</span>, <span class="function">len</span>(arr)):<br>
                            &nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;<span class="keyword">if</span> arr[i] == arr[j]:<br>
                            &nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;<span class="keyword">return</span> <span class="keyword">True</span><br><br>
                            
                            <span class="comment"># O(n) - Linear Time (Fast)</span><br>
                            <span class="keyword">def</span> <span class="function">find_duplicates_fast</span>(arr):<br>
                            &nbsp;&nbsp;seen = <span class="function">set</span>()<br>
                            &nbsp;&nbsp;<span class="keyword">for</span> num <span class="keyword">in</span> arr:<br>
                            &nbsp;&nbsp;&nbsp;&nbsp;<span class="keyword">if</span> num <span class="keyword">in</span> seen:<br>
                            &nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;<span class="keyword">return</span> <span class="keyword">True</span><br>
                            &nbsp;&nbsp;&nbsp;&nbsp;seen.<span class="function">add</span>(num)<br>
                            &nbsp;&nbsp;<span class="keyword">return</span> <span class="keyword">False</span>
                        </div>
