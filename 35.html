# Local variable example
def calculate_area(length, width):
  # Local variable
  area = length * width
  print(f"Inside function: {area}")
  return area

# Area is not accessible outside
result = calculate_area(5, 3)
print(result) # 15
# print(area) would cause NameError

# Global variable defined outside functions
counter = 0

# Variable lifetime example
def demonstrate_lifetime():
  # Local variable created
  local_var = "I exist only inside"
  print(local_var)
  # Function ends, local_var is destroyed

# Using id() to check object existence
demonstrate_lifetime()
try:
    print("Local var still exists?", local_var is not None)
except NameError:
    print("Local var no longer exists")
