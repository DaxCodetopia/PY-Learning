        <section id="why-testing">
            <h2>1. Why Do We Test?</h2>
            <p>Software testing is not just about finding bugs; it's about ensuring reliability, user satisfaction, and business continuity. Here are the core reasons:</p>
            <ul>
                <li><strong>To identify defects:</strong> Catching errors early reduces the cost of fixing them significantly.</li>
                <li><strong>To ensure quality:</strong> Verifies that the software meets the specified requirements and standards.</li>
                <li><strong>To provide security:</strong> Helps in identifying vulnerabilities and security risks.</li>
                <li><strong>To improve user experience:</strong> Ensures the application is intuitive and performs well under load.</li>
            </ul>
        </section>

        // --- Theme Toggle Logic ---
        const themeToggleBtn = document.getElementById('theme-toggle');
        const htmlElement = document.documentElement;
        
        // Check local storage for saved theme
        const savedTheme = localStorage.getItem('theme');
        const systemPrefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches;

        // Set initial theme
        if (savedTheme) {
            htmlElement.setAttribute('data-theme', savedTheme);
            updateButtonIcon(savedTheme);
        } else if (systemPrefersDark) {
            htmlElement.setAttribute('data-theme', 'dark');
            updateButtonIcon('dark');
        } else {
            htmlElement.setAttribute('data-theme', 'light');
            updateButtonIcon('light');
        }

        // Handle click event
        themeToggleBtn.addEventListener('click', () => {
            const currentTheme = htmlElement.getAttribute('data-theme');
            const newTheme = currentTheme === 'light' ? 'dark' : 'light';
            
            htmlElement.setAttribute('data-theme', newTheme);
            localStorage.setItem('theme', newTheme);
            updateButtonIcon(newTheme);
        });

        function updateButtonIcon(theme) {
            if (theme === 'dark') {
                themeToggleBtn.textContent = '☀️'; // Sun icon to switch back to light
            } else {
                themeToggleBtn.textContent = '🌙'; // Moon icon to switch to dark
            }
        }

        // --- Original Report Generation Logic ---
