# String manipulation functions
def reverse_string(s):
  return s[::-1]

def count_vowels(s):
  vowels = "aeiouAEIOU"
  count = 0
  for char in s:
    if char in vowels:
        count += 1
  return count

def is_palindrome(s):
  # Remove non-alphanumeric characters and convert to lowercase
  clean_s = "".join(c for c in s if c.isalnum())
  clean_s = clean_s.lower()
  return clean_s == clean_s[::-1]

# Word processing
def word_frequency(text):
  words = text.lower().split()
  freq = {}
  for word in words:
    if word in freq:
        freq[word] += 1
    else:
        freq[word] = 1
  return freq

Recursive Functions
Create functions that call themselves to solve problems
# Factorial using recursion
def factorial(n):
  # Base case
  if n <= 1:
    return 1
  # Recursive case
  return n * factorial(n-1)

# Fibonacci sequence
def fibonacci(n):
  # Return first n numbers in sequence
  if n <= 0:
    return []
  elif n == 1:
    return [0]
  else:
    seq = fibonacci(n-1)
    seq.append(seq[-1] + seq[-2])
    return seq

# Binary search using recursion
def binary_search(arr, target, low=0, high=None):
  if high is None:
    high = len(arr) - 1
  mid = (low + high) // 2
  if arr[mid] == target:
    return mid
  elif arr[mid] < target:
    return binary_search(arr, target, mid+1, high)
  else:
    return binary_search(arr, target, low, mid-1)

# Input validation functions
def validate_email(email):
  # Basic email validation
  if "@" not in email or "." not in email.split("@")[1]:
    return False, "Invalid email format"
  return True, "Valid email"

# Password strength checker
def check_password_strength(password):
  if len(password) < 8:
    strength = "Weak"
  elif len(password) < 12:
    strength = "Medium"
  elif len(password) >= 12 and any(c.isupper() for c in password) and any(c.islower() for c in password) and any(c.isdigit() for c in password):
    strength = "Strong"
  else:
    strength = "Very Weak"
  return strength
