# Debugging a function with errors
def calculate_average(numbers):
  # Debug: Check if list is empty
  print(f"Input: {numbers}")
  if not numbers:
    return 0

  # Debug: Check sum and count
  total = sum(numbers)
  count = len(numbers)
  print(f"Sum: {total}, Count: {count}")

  # Original error: Missing parentheses
  # return total / count # This would cause a TypeError
  # Fixed: Add parentheses
  return total / count

# Test the function
result = calculate_average([10, 20, 30])
print(f"Average: {result}")
