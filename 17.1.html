The open() Function
# Basic syntax: open(filename, mode)
file = open("example.txt", "r")

# Using with statement (recommended)
with open("example.txt", "r") as file:
  content = file.read()
  print(content)
Mode Description
'r' Read mode (default)
'w' Write mode (overwrites)
'a' Append mode
'r+' Read and Write mode
'b' Binary mode

Writing to a File
# Writing (overwrites existing file)
with open("example.txt", "w") as file:
  file.write("Hello, Python!\n")
  file.write("File handling is easy.\n")

# Writing multiple lines
lines = ["Line 1\n", "Line 2\n", "Line 3\n"]
with open("example.txt", "w") as file:
  file.writelines(lines)

Reading from a File
# Reading the entire file
try:
  with open("example.txt", "r") as file:
    content = file.read()
    print(content)
except FileNotFoundError:
  print("The file was not found.")

# Reading line by line (memory efficient)
with open("example.txt", "r") as file:
  for line in file:
    print(line.strip()) # removes newline

# Reading all lines into a list
with open("example.txt", "r") as file:
  lines = file.readlines()
  print(lines)

Appending to a File
# Appending to an existing file
with open("example.txt", "a") as file:
  file.write("This line is appended.\n")

# Appending with read/write mode
with open("example.txt", "r+") as file:
  content = file.read()
  file.seek(0, 2) # Move to end of file
  file.write("\nNew content at the end")

# Binary mode for non-text files
with open("image.jpg", "rb") as img_file:
  data = img_file.read()
  print(len(data))
check_circle
'a' mode adds content to end of file
check_circle
Use 'b' with other modes for binary files
